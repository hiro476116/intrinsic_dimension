% --- PDF p.04 | SDEを用いた生成 ---
\begin{frame}{拡散モデルについて : SDEを用いた生成}
    $\{W_t\}_{t\in[0,T]}$ をブラウン運動として, 次のSDEを考える.
    \begin{align*}
        dY_t=b(t,Y_t)dt+\sigma(t,Y_t)dW_t\quad (t\in[0,T])
    \end{align*}
    上記のSDEの解 $\{Y_t\}_{t\in[0,T]}$ が次の2つの性質を満たすように, $b$ と $\sigma$ を設計できるかということを考える.
    \begin{enumerate}
      \item 初期分布 $\mathcal{L}(Y_0)$ が, 容易にサンプルが得られる分布である (正規分布, 一様分布など)
      \item 最終時刻の分布が $P_{\text{data}}$ と等しい. つまり, $\mathcal{L}(Y_T)=P_{\text{data}}$ .
    \end{enumerate}

    \vspace{0.15cm}
    \begin{center}
    \begin{tikzpicture}[
      node distance=5.2cm,
      box/.style={
        draw,
        rounded corners,
        minimum width=2.8cm,

      },

        -{Latex[length=3mm]},
        thick
      }

    \node[box] (normal) {
      $Y_0$\\

      $Y_T\sim P_{\text{data}}$\\

\end{frame}
% --- PDF p.05 | Ornstein--Uhlenbeck process ---
\begin{frame}{拡散モデルについて : Ornstein Uhlenbeck process}
    \begin{block}{Ornstein Uhlenbeck process}
      次のSDEの解 $\{X_t\}_{t\in[0,T]}$ をOrnstein–Uhlenbeck processという.
      \begin{equation*}
        dX_t=-X_tdt+\sqrt{2}dW_t \quad (t\in[0,T])
      \end{equation*}
    \end{block}
      Ornstein–Uhlenbeck processは明示的に解くことができ, $X_0$ と独立な標準正規分布に従う確率変数 $Z$ が存在して,
      \begin{equation*}
        X_t=e^{-t}X_0+\sqrt{2}\int_{0}^{t}e^{-(t-s)}dW_s=e^{-t}X_0+\sqrt{1-e^{-2t}}Z
      \end{equation*}
      と書ける. これにより, $T$ を十分大きくとると $X_T$ の分布は標準正規分布 $\mathcal{N}(0,I_d)$ に近づいていく.

      $\mathcal{L}(X_0)=P_{\text{data}}$ とすると, $X$ は $P_\text{data}$ から時間経過とともに
      $\mathcal{N}(0,I)$ へ近づいていく過程とみなせる.
    \begin{center}
    \begin{tikzpicture}[
      node distance=5.2cm,
      box/.style={
        draw,
        rounded corners,

        minimum height=1.15cm,

        -{Latex[length=3mm]},

    \node[box] (normal) {
      $\mathcal{L}(X_T)\approx \mathcal{N}(0,I_d)$
    };

    \node[box, right=of normal] (data) {
      $\mathcal{L}(X_0)= P_{\text{data}}$\\
      \small データ分布
    };

    \end{tikzpicture}

\end{frame}
% --- PDF p.06 | 拡散過程の時間反転公式 ---
\begin{frame}{拡散モデルについて : 拡散過程の時間反転公式}
    $\{X_t\}_{t\in[0,T]}$ をOrnstein–Uhlenbeck processとして, $0<t\leqq T$ のとき, $X_t$ の密度を $p_t$ とする.
    \begin{theorem}[\cite{anderson82,haussmann_pardoux86}]
      $\{Y_t\}_{t\in[0,T]}$ を次のSDEの解とする.
      \begin{equation}
        \left\{ \,
            \begin{aligned}
            & dY_t = (Y_t+2\nabla \log{p_{T-t}(Y_t)})dt+\sqrt{2}dW_t \quad (0< t < T) \\
            & Y_0 \stackrel{\mathrm{d}}{=} X_T
            \end{aligned}
        \right.
      \end{equation}
      このとき, 任意の $t\in[0,T]$ に対して, $Y_t\stackrel{\mathrm{d}}{=}X_{T-t}$ が成り立つ.
    \end{theorem}
    $\{X_t\}_{t\in[0,T]}$ の時間反転過程 $\{Y_t\}_{t\in[0,T]}$ は, $\mathcal{L}(X_T)\approx \mathcal{N}(0,I_d)$ から時間の経過とともに
        $P_{\text{data}}$ へ近づいていく確率過程となる.
    \vspace{-0.5em}
    \begin{center}
    \begin{tikzpicture}[
      scale=0.92,

        draw,

      },
      arrow/.style={
        -{Latex[length=3mm]},
        thick
      }
    ]

      \node[box] (normal) {
          $\mathcal{L}(Y_0)=\mathcal{L}(X_T)$\\
          $\approx \mathcal{N}(0,I_d)$
      };

    \node[box, right=of normal] (data) {
      $\mathcal{L}(Y_T)=\mathcal{L}(X_0)$\\
      $\sim P_{\text{data}}$
    };

    \draw[arrow]

\end{frame}


% --- PDF p.07 | 3つの誤差の発生源 ---
\begin{frame}{拡散モデルについて : 3つの誤差の発生源}
    \begin{equation*}
        \left\{ \,
            \begin{aligned}
            & dY_t = (Y_t+2\nabla \log{p_{T-t}(Y_t)})dt+\sqrt{2}dB_t \quad (0\leqq t \leqq T) \\
            & Y_0 \stackrel{\mathrm{d}}{=} X_T\approx \mathcal{N}(0,I_d)
            \end{aligned}
        \right.
    \end{equation*}
      上記のSDEをシミュレーションすると, $Y_T\sim P_{\text{data}}$ からのサンプリングが得られる. しかし, 現実的には以下の3つの点が問題となる.
      \begin{enumerate}
        \item 初期分布 $\mathcal{L}(X_T)$ からの厳密なサンプリングが得られない.
        \item スコア関数 $\nabla \log{p_{T-t}(Y_t)}$ が明示的に得られない.
        \item SDEを連続時間でシミュレーションできない.
      \end{enumerate}
      拡散モデルでは, それぞれの問題点に対して次のように対応する.
      \begin{enumerate}
        \item 初期分布を標準正規分布 $\mathcal{N}(0,I_d)$ にする.
        \item スコア関数 $\nabla\log{p_{t}(y)}$ を推定量 $\hat{s}(t,y)$ に置き換える.
        \par\vspace{0.15em}
        \hspace*{1.4em}\(\rightarrow\)\hspace{0.4em}%
        \begin{minipage}[t]{0.82\linewidth}
          \small
          手元にあるデータ $x_1,\ldots,x_n\sim P_{\text{data}}$ を使い,
          スコア関数との平均二乗誤差が小さくなるようにニューラルネットワークで近似できる.
        \end{minipage}
        \par\vspace{0.15em}
        \item 時間に関して離散化する.
      \end{enumerate}
\end{frame}

% --- PDF p.08 | 収束解析の目的 ---
\begin{frame}{拡散モデルの収束解析について}
      実際にシミュレーションして得られるデータの分布を $\hat{P}_\text{data}$ とする.
      このとき, 確率測度の近さを測る指標 $\rho$ を用いて, $\rho(P_\text{data},\hat{P}_\text{data})$ を評価することを,
    \textcolor{red}{拡散モデルの収束解析} と呼ぶ.


    指標 $\rho$ として用いられるものは, 主に次の2種類である.
    {\large
      \begin{enumerate}
        \item Kullback-Leibler divergence
        \begin{itemize}
          \item 距離の公理は満たさないが, 広く用いられている指標
        \end{itemize}
        \item 2-Wasserstein distance
        \begin{itemize}
          \item 距離の公理を満たす.
          \item 拡散モデルの収束解析においては, Kullback-Leibler divergenceよりも解析が困難である場合が多い.
          \item 画像の類似度において, 人間の知覚との相関があると言われている.
        \end{itemize}
      \end{enumerate}
    }

\end{frame}
% --- PDF p.09 | Kullback--Leibler divergence ---
{
    \setbeamerfont{frametitle}{size=\large}
\begin{frame}{拡散モデルの収束解析について : Kullback-Leibler divergence}
      \small
      \footnotesize
      \begin{definition}
        2つの確率測度 $P, Q$ のKullback-Leibler divergence $D_{\text{KL}}(P|Q)$ を, 次のように定義する.
        \begin{equation*}
          D_{\text{KL}}(P\mid Q):=        \left\{ \,
            \begin{aligned}
            & \mathbb{E}_P\left[\log{\frac{dP}{dQ}}\right] \quad \bigl(P\ll Q\bigr) \\
            & \infty \quad \quad \quad\quad \quad \quad(\text{otherwise})
            \end{aligned}
        \right.
        \end{equation*}
      \end{definition}
      \small
      \vspace{-0.65em}
      \begin{theorem}[\cite{benton_et_al_2024}]
        ある定数 $C>0$ が存在して, 次の不等式が成立する.
        \vspace{-0.35em}
        \begin{equation*}
          D_{\mathrm{KL}}\bigl(P_{\text{data}}\mid \hat{P}_{\text{data}}\bigr)
          \leqq C\left(\underbrace{\frac{2dT^2}{N}}_{\text{離散化誤差}}+\underbrace{\epsilon_{\text{score}}}_{\text{スコア推定誤差}}+\underbrace{\left(d+\E[\|X_0\|_2^2]\right)e^{-2T}}_{\text{初期分布誤差}}\right)
        \end{equation*}
        \vspace{-0.2em}
        \noindent
        $t_n=\frac{nT}{N}\quad(n=0,\ldots,N)\quad\epsilon_{\text{score}}=\frac{T}{N}\sum\nolimits_{n=1}^{N}
          \mathbb{E}\!\left[
            \left\|\nabla\log p_{t_n}(X_{t_n})
            -\hat{s}(t_n,X_{t_n})\right\|^2
          \right]$
        \par\vspace{0em}
        \noindent{\scriptsize
          $N$: 時間離散化のステップ数\quad
          $d$: データの次元\quad
          $T$: SDEの終端時刻
        }
      \end{theorem}
\end{frame}
  }
% --- PDF p.10 | Wasserstein distance ---
\begin{frame}{拡散モデルの収束解析について : Wasserstein distance}
    \small
    \begin{definition}
      $\mathbb{R}^d$ 上の確率測度 $P,Q$ は有限な2次モーメントをもつ, すなわち
      $\int\|x\|_2^2\,dP(x)+\int\|y\|_2^2\,dQ(y)<\infty$ と仮定する. このとき,
      $\Pi(P,Q)=\{\pi\text{ : }\mathbb{R}^d\times\mathbb{R}^d\text{上の確率測度}
      \mid\pi(\cdot,\mathbb{R}^d)=P,\pi(\mathbb{R}^d,\cdot)=Q\}$とする.
      このとき, $P,Q$の2-Wasserstein distance $W_2(P,Q)$を次のように定義する.
      \begin{equation*}
        W_2(P,Q)=\inf_{\pi\in\Pi(P,Q)}
        \left(\int_{\mathbb{R}^d\times\mathbb{R}^d}|x-y|^2d\pi(x,y)\right)^{\frac{1}{2}}.
      \end{equation*}
    \end{definition}
    \vspace{-0.4em}
    \begin{theorem}[\cite{koike_2026_wasserstein}, 定理 3.3]
      $t_n=\frac{nT}{N}\ (n=0,\ldots,N)$ とする.
      適切な正則性条件のもとで, ある定数
      $C_L>0$ が存在して, 次の不等式が成立する.
      \vspace{-0.4em}
      \begin{equation*}
        W_2\bigl(P_\text{data},\hat{P}_\text{data}\bigr)
        \leqq C_L\Biggl(
          \underbrace{\frac{\sqrt{d}T}{N}}_{\text{離散化誤差}}
          +\underbrace{\sqrt{\epsilon_{\mathrm{score}}}}_{\text{スコア推定誤差}}
          +\underbrace{e^{-2T}\sqrt{d+\E[\|X_0\|_2^2]}}_{\text{初期分布誤差}}
        \Biggr).
      \end{equation*}
      \vspace{-0.2em}
      \noindent{\scriptsize
        $N$: 時間離散化のステップ数\quad
        $d$: データの次元\quad
        $T$: SDEの終端時刻
      }
    \end{theorem}
\end{frame}
% --- PDF p.11 | データの内在次元 ---
\begin{frame}{拡散モデルの収束解析について : データの内在次元}
    現実世界のデータ分布はごく少数の変数で記述されると考えられている. これは\textcolor{red}{多様体仮説}と呼ばれ, 機械学習の成功の基盤と考えられている (\cite{pope_et_al_2021}).
    \begin{center}
      \includegraphics[
        width=0.60\linewidth,
        trim=0 0 350bp 0,
        clip
      ]{\detokenize{スクリーンショット 2026-08-08 16.19.15.png}}
    \end{center}
    データ分布を記述する上で必要な変数の数を, データ分布の\textcolor{red}{内在次元} と呼ぶ.

    \cite{pope_et_al_2021} は, 代表的な画像のデータセットに対する内在次元推定を行い, 画像は $224\times 224 \times 3 = 15028$ の画素値を持つが, 内在次元は $26\sim 43$ 程度であるという結果を得ている.
\end{frame}
% --- PDF p.12 | 内在次元を用いた既存結果 ---
\begin{frame}{拡散モデルの収束解析について : データの内在次元}
    データの次元 $d$ ではなく, 内在次元 $k$ を使った収束解析はKullback-Leibler divergence において行われており, 以下の結果が得られている.
    \begin{theorem}[\cite{huang_wei_chen_2026}]
      $t_n=\frac{nT}{N}\ (n=0,\ldots,N)$ とする. 内在次元に対する適切な仮定のもと,
      ある定数 $C>0$ が存在して, 次の不等式が成立する.
      \vspace{-0.5em}
      \begin{equation*}
        \begin{aligned}
          KL\left(P_{\text{data}}\mid\hat{P}_\text{data}\right)
          \leqq C\Biggl(&
          \underbrace{\frac{T^2}{N}k\log k}_{\text{離散化誤差}}
          +\underbrace{\epsilon_{\text{score}}}_{\text{スコア推定誤差}}
          +\underbrace{\left(d+\E[\|X_0\|_2^2]\right)e^{-2T}}_{\text{初期分布誤差}}
          \Biggr).
        \end{aligned}
      \end{equation*}
      \vspace{-0.2em}
      \noindent{\scriptsize
        $N$: 時間離散化のステップ数\quad
        $d$: データの次元\quad
        $T$: SDEの終端時刻\quad
        $k$: 内在次元
      }
    \end{theorem}
\end{frame}
% --- PDF p.13 | 主結果の比較 ---
\begin{frame}[t]{主結果}
  \vspace{1.5cm}
  \centering
  \renewcommand{\arraystretch}{1.4}
  \setlength{\tabcolsep}{0.35em}
  \begin{tabular}{
    >{\centering\arraybackslash}m{0.08\textwidth}|
    >{\centering\arraybackslash}m{0.39\textwidth}|
    >{\centering\arraybackslash}m{0.42\textwidth}}
    \hline
    & {\large\textbf{KL divergence}}
    & {\large\textbf{2-Wasserstein distance}} \\
    \hline
    {\large $d$次元}
    & {\small
      $\displaystyle
        \lesssim
          \frac{dT^2}{N}
          +\epsilon_{\mathrm{score}}
          +de^{-2T}
      $
      \par\vspace{0.25em}
      {\footnotesize Benton et al. (2024)}}
    & {\small
      $\displaystyle
      \begin{aligned}
        \lesssim
          \frac{\sqrt d\,T}{N}

% --- PDF p.14 | 主結果の定理 ---
\begin{frame}{主結果 : 定理}
      \begin{theorem}
        $t_n=\frac{nT}{N}\ (n=0,\ldots,N)$ とする.
        このとき, 正則性条件と, 内在次元に対する適切な仮定のもと, ある定数$C>0$ が存在して, 次が成立する.
        \begin{align*}
            W_2(P_{\text{data}}, \hat{P}_\text{data})
            &\leqq C\Biggl(
            \underbrace{\sqrt{\frac{Tk\log{k}}{N}}}_{\text{離散化誤差}}
            +\underbrace{\sqrt{\epsilon_{\text{score}}}}_{\text{スコア推定誤差}}
            \notag\\[-0.1em]
            &\phantom{\leqq C\Biggl(}+
            \underbrace{e^{-2T}\left(\sqrt{\E[\|X_0\|_2^2]}+e^{-T}\sqrt d\right)}_{\text{初期分布誤差}}
            \Biggr)
        \end{align*}
      \vspace{-0.2em}
      \noindent{\scriptsize
        $N$: 時間離散化のステップ数\quad
        $d$: データの次元\quad
        $T$: SDEの終端時刻\quad
        $k$: 内在次元
      }
      \end{theorem}
\end{frame}


% --- PDF p.15 | Covering number ---
\begin{frame}[t]{設定 : データの内在次元の仮定}
      \begin{definition}
        $A\subset \mathbb{R}^d$ に対する, スケール $\epsilon_0$ における \textcolor{red}{covering number} $N_{cover}(A,\|\|_2,\epsilon_0)$ を, 次の関係を満たす $x_1,\cdots, x_s \in \mathbb{R}^d$
        が存在するような最小の自然数 $s$ として定義する.
        \begin{equation*}
          A\subset \bigcup_{i=1}^s B(x_i,\epsilon_0)
        \end{equation*}
        また, $\log{N_{cover}(A,\|\|_2,\epsilon_0)}$ を \textcolor{red}{metric entropy} という.
      \end{definition}
      \vspace{-0.1em}
      \begin{block}{例}
        \footnotesize
        \begin{itemize}
          \item $k$次元線形部分空間内の有界集合を $A\subset B(0,R)$ としたとき, $\log N_{cover}(A,\|\|_2,\epsilon_0)
          \leqq k\log\left(1+\frac{2R}{\epsilon_0}\right)$
          \item 正則なコンパクト$k$次元多様体を $M$ としたとき,
          $\log N_{cover}(M,\|\|_2,\epsilon_0)\leqq\mathcal O\!\left(k\log\left(\frac{1}{\epsilon_0}\right)\right)$
        \end{itemize}
      \end{block}
\end{frame}

% --- PDF p.16 | 内在次元の仮定 ---
\begin{frame}{設定 : データの内在次元の仮定}
      $P_{data}$ の サポートを $\chi_{data}$ とする.
      $\chi_{data}$ に対して, 次の2つの仮定を置く.
      \begin{assumption}
        $C_0 > 0$ を十分大きな定数とする. $\epsilon_0=k^{-C_0}$ に対して, ある $C_{cover}>0$ が存在して,
        \begin{equation}
          \log{N_{cover}(\chi_{data},\|\|_2,\epsilon_0)}\leqq C_{cover}k\log{\frac{1}{\epsilon_0}}
        \end{equation}
        を満たすと仮定する. このとき $k$ を $\chi_{data}$ の\textcolor{red}{内在次元}という.
      \end{assumption}
      \begin{assumption}
        ある定数 $C_R>0$ が存在して, 次が成立すると仮定する.
        \begin{equation}
          \sup\limits_{x\in \chi_{data}}\|x\|_2\leqq k^{C_R}
        \end{equation}
      \end{assumption}
\end{frame}

% --- PDF p.17 | Tweedieの公式 ---
\begin{frame}[t]{設定：Tweedieの公式}
  \begin{theorem}[Tweedieの公式]
    $X$ を $\mathbb{R}^d$ 値の任意の確率変数とし, $Z\sim \mathcal{N}(0,I_d)$ を $X$ と独立とする. この時, $Y=X+\sqrt{t}Z$ として, $Y$ の密度関数を $p$ とすると, 次の関係が成り立つ.
    \begin{equation*}
      \mathbb{E}[X\mid Y=y]=y+t\nabla\log p(y)
    \end{equation*}
  \end{theorem}
  ある $Z\sim \mathcal{N}(0,I_d), Z\perp X_0$ が存在して $X_t=e^{-t}X_0+\sqrt{1-e^{-2t}}Z$ と書けることから,
  Tweedieの公式により
  \begin{equation*}
    e^{-t}\mathbb{E}[X_0\mid X_t=x]
    =x+(1-e^{-2t})\nabla\log{p_t(x)}
  \end{equation*}
  となる. よって,
  \begin{equation*}
    \nabla\log{p_t(x)}
    =\frac{e^{-t}\mathbb{E}[X_0\mid X_t=x]-x}{1-e^{-2t}}
  \end{equation*}
  となる.

\end{frame}

% --- PDF p.18 | Tweedieの公式の適用 ---
\begin{frame}[t]{設定：Tweedieの公式}
  \small
  いま, $Y_t=X_{T-t} \quad (t\in[0,T])$ とすると,
  \begin{align*}
    \nabla\log{p_{T-t}(y)}
    &=\frac{e^{-(T-t)}\mathbb{E}[X_0\mid X_{T-t}=y]-y}
            {1-e^{-2(T-t)}}\\
    &=\frac{e^{-(T-t)}\mathbb{E}[Y_T\mid Y_t=y]-y}
            {1-e^{-2(T-t)}}
  \end{align*}
  となる. よって, $\{Y_t\}_{t\in[0,T]}$ が従うSDEは次のように変形できる.
      \begin{equation*}
        \left\{ \,
            \begin{aligned}
            & dY_t = (Y_t+2\nabla \log{p_{T-t}(Y_t)})dt+\sqrt{2}dW_t \\
            & Y_0 \stackrel{\mathrm{d}}{=} X_T
            \end{aligned}
        \right.
      \end{equation*}

      \vspace{-1.2em}
      \centerline{\large$\Downarrow$}
      \vspace{-1.2em}

      \begin{equation*}
        \left\{ \,
            \begin{aligned}
            dY_t &={} \left\{\left(1-\frac{2}{1-e^{-2(T-t)}}\right)Y_t
              +\frac{2e^{-(T-t)}}{1-e^{-2(T-t)}}\mathbb{E}[Y_T\mid Y_t]\right\}dt+\sqrt{2}dW_t \\
            Y_0 &\stackrel{\mathrm{d}}{=} X_T
            \end{aligned}
        \right.
      \end{equation*}

\end{frame}

% --- PDF p.19 | 実際にシミュレーションするSDE ---
\begin{frame}[t]{設定：実際にシミュレーションするSDE}
  時間区間 $[0,T]$ を $N$ 等分し, 離散化時刻を
    $t_n=\frac{nT}{N}\quad(n=0,\ldots,N)$
  とする.また, Tweedieの公式
  \begin{equation*}
    \mathbb{E}[Y_T\mid Y_t=y]
    =e^{T-t}\left\{y+(1-e^{-2(T-t)})\nabla\log{p_{T-t}(y)}\right\}
  \end{equation*}
  に対して,
  \begin{equation*}
    \hat{\mu}_r(y)=e^r\left\{y+(1-e^{-2r})\hat{s}(r,y)\right\}
  \end{equation*}
  として, 各 $t\in[t_n,t_{n+1}]$ において実際にシミュレーションするSDEを次のように定める.
      {\small
        \begin{equation*}
          \left\{ \,
              \begin{aligned}
              d\hat{Y}_t &={} \left\{\left(1-\frac{2}{1-e^{-2(T-t)}}\right)\hat{Y}_t
                +\frac{2e^{-(T-t)}}{1-e^{-2(T-t)}}\hat{\mu}_{T-t_n}(\hat{Y}_{t_n})\right\}dt+\sqrt{2}dW_t \\
              \hat{Y}_0 &\sim \mathcal{N}(0,I_d)
              \end{aligned}
          \right.
        \end{equation*}
      }
\end{frame}


% --- PDF p.20 | 証明の概要 ---
\begin{frame}[t]{証明の概要}
      Ornstein Uhlenbeck process $\{X_t\}_{t\in [0,T]}$ に対して,
      $\{Y_t\}_{t\in[0,T]}$ を $Y_t=X_{T-t}$ としてとる.
      各 $t\in[t_n,t_{n+1}]$ において
      {\small
        \begin{equation*}
          \left\{ \,
              \begin{aligned}
              d\hat{Y}_t &={} \left\{\left(1-\frac{2}{1-e^{-2(T-t)}}\right)\hat{Y}_t
                +\frac{2e^{-(T-t)}}{1-e^{-2(T-t)}}\hat{\mu}_{T-t_n}(\hat{Y}_{t_n})\right\}dt+\sqrt{2}dW_t \\
              \hat{Y}_0 &=Z\sim \mathcal{N}(0,I_d)
              \end{aligned}
          \right.
        \end{equation*}
      }
      の解を $\{\hat{Y}_t\}_t\in[0,T]$ とする.
  このときWasserstein 距離の定義から,
        \begin{equation*}
          W_2(P_{\text{data}},\hat{P}_{\text{data}})
          \leqq
          \E\left[\|Y_{t_N}-\hat{Y}_{t_N}\|_2^2\right]^{\frac{1}{2}}.
        \end{equation*}
  となる.
\end{frame}

% --- PDF p.21 | 誤差の分解 ---
\begin{frame}[t]{証明の概要 : 誤差の分解}
  平均二乗誤差をとり, 適切に三角不等式やGronwall型の不等式を用いると, ある定数 $C_1>0$ に対して,
      {
      \begin{align*}
        \E\left[\|Y_{t_N}-\hat{Y}_{t_N}\|_2^2\right]^{\frac{1}{2}}
        &\leqq
        C_1 \Bigg[
            {\underbrace{\E\left[\|Y_0-\hat{Y}_0\|_2^2\right]^{\frac{1}{2}}}_{\text{初期分布誤差}}}
            \notag\\[-0.2em]&\quad+
            {\underbrace{\begin{aligned}
              &\left(\sum_{n=0}^{N-1}\int_{t_n}^{t_{n+1}}
              \frac{e^{-2(T-t)}}{\left(1-e^{-2(T-t)}\right)^2}\right.\\[-0.2em]
              &\left.\quad\times\E\left[
                \|\mathbb{E}\left[Y_T\mid Y_t\right]-\mathbb{E}\left[Y_T\mid Y_{t_n}\right]\|_2^2
              \right]dt\right)^{\frac{1}{2}}
            \end{aligned}}_{\text{離散化誤差}}}
            \notag\\[-0.2em]&\quad+
          {\underbrace{\left(\frac{T}{N}\sum\limits_{n=1}^{N}
          \mathbb{E}\left[
            \left\|\nabla\log p_{t_n}(X_{t_n})
            -\hat{s}(t_n,X_{t_n})\right\|^2
          \right]
          \right)^\frac{1}{2}}_{\text{スコア推定誤差}}}
        \Bigg].
      \end{align*}
      }
  のように, 誤差を分解することができる.
\end{frame}
% --- PDF p.22 | 初期分布誤差とスコア推定誤差 ---
\begin{frame}{証明の概要：初期分布誤差, スコア推定誤差を評価}
      初期分布の取り方より,
      \begin{align*}
        \mathbb{E}\left[\left|Y_0-\hat{Y}_0\right|^2\right]&=\mathbb{E}\left[\|e^{-T}X_0+\sqrt{1-e^{-2T}}Z-Z\|^2\right]\\
        &=e^{-2T}\mathbb{E}\left[\|X_0\|^2\right]+(1-\sqrt{1-e^{-2T}})^2d\\
        &\leqq e^{-2T}\mathbb{E}\left[\|X_0\|^2\right]+e^{-4T}d
      \end{align*}
      と評価ができる.
      また, 仮定より,
      \begin{align*}
        \left(\frac{T}{N}\sum\limits_{n=1}^{N}
          \mathbb{E}\left[
            \left\|\nabla\log p_{t_n}(X_{t_n})
            -\hat{s}(t_n,X_{t_n})\right\|^2
          \right]
          \right)^\frac{1}{2}=\sqrt{\epsilon_{\text{score}}}
      \end{align*}
      と評価できる.
\end{frame}
% --- PDF p.23 | 離散化誤差の恒等式 ---
\begin{frame}{証明の概要：離散化誤差を評価}
      \begin{lemma}
        \begin{align*}\label{eq:posterior-mean-increment}
          &\E\left[
            \|\mathbb{E}\left[Y_T\mid Y_t\right]
            -\mathbb{E}\left[Y_T\mid Y_s\right]\|_2^2
          \right]\\[-0.1em]
          &\quad=
          \E\left[\operatorname{Tr}\left(\operatorname{Cov}\left(Y_T\mid Y_s\right)\right)\right]
          -\E\left[\operatorname{Tr}\left(\operatorname{Cov}\left(Y_T\mid Y_t\right)\right)\right].
      \end{align*}
      \end{lemma}
      この補題により, 離散化誤差は次のように変形できる.
      \begin{align*}
        &\sum_{n=0}^{N-1}\int_{t_n}^{t_{n+1}}
                \frac{e^{-2(T-t)}}{\left(1-e^{-2(T-t)}\right)^2}\E\left[
              \|\mathbb{E}\left[Y_T\mid Y_t\right]-\mathbb{E}\left[Y_T\mid Y_{t_n}\right]\|_2^2
          \right]dt\\
        &=\sum_{n=0}^{N-1}\int_{t_n}^{t_{n+1}}
                \frac{e^{-2(T-t)}}{\left(1-e^{-2(T-t)}\right)^2}\Bigl(\E\left[Tr\left(Cov\left(Y_T\mid Y_{t_n}\right)\right)\right]\\
          &\quad\quad\quad\quad\quad\quad\quad\quad\quad \quad \quad \quad\quad
          -\E\left[Tr\left(Cov\left(Y_T\mid Y_t\right)\right)\right]\Bigr)dt
      \end{align*}
\end{frame}

% --- PDF p.24 | 離散化誤差の評価 ---
\begin{frame}{証明の概要 : 離散化誤差を評価}
      \begin{lemma}
        ある定数 $C_2>0$ が存在して,
        \begin{align*}
          &\sum_{n=0}^{N-1}\int_{t_n}^{t_{n+1}}
          \frac{e^{-2(T-t)}}{\left(1-e^{-2(T-t)}\right)^2}\\[-0.1em]
          &\quad\times\Bigl(
          \E\left[\operatorname{Tr}\left(\operatorname{Cov}\left(Y_T\mid Y_{t_n}\right)\right)\right]
          -\E\left[\operatorname{Tr}\left(\operatorname{Cov}\left(Y_T\mid Y_t\right)\right)\right]
          \Bigr)\,dt\\[0.2em]
          &\leqq C_2\frac{T}{N}\log N_{\mathrm{cover}}\left(
          \chi_{\mathrm{data}},\|\cdot\|_2,k^{-C_0}
          \right).
        \end{align*}
      \end{lemma}
      また, 仮定よりある定数 $C_3>0$ が存在して,
      \begin{align*}
        C_2\frac{T}{N}\log N_{\mathrm{cover}}\left(
          \chi_{\mathrm{data}},\|\cdot\|_2,k^{-C_0}\right)\leqq \frac{T}{N}C_3 k\log{k}
      \end{align*}
      となる.
\end{frame}

% --- PDF p.25 | 最後の評価 ---
\begin{frame}{証明の概要：最後の評価}
      前頁までの評価をまとめると,
        \begin{align*}
            W_2(P_{\text{data}}, \hat{P}_\text{data})
            &\leqq C\Biggl(
            \underbrace{\sqrt{\frac{Tk\log{k}}{N}}}_{\text{離散化誤差}}
            +\underbrace{\sqrt{\epsilon_{\text{score}}}}_{\text{スコア推定誤差}}
            \notag\\[-0.1em]
            &\phantom{\leqq C\Biggl(}+
            \underbrace{e^{-2T}\left(\sqrt{\E[\|X_0\|_2^2]}+e^{-T}\sqrt d\right)}_{\text{初期分布誤差}}
            \Biggr)
        \end{align*}
      が得られる.
\end{frame}
