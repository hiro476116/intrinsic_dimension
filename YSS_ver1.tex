  \begin{frame}{拡散モデルについて : 拡散モデルの原理}
    \begin{itemize}
      \item 拡散モデルでは, 確率微分方程式(SDE)を用いてデータを生成する.
    \end{itemize}
    \begin{align}
        dY_t=b(t,Y_t)dt+\sigma(t,Y_t)dW_t\quad (t\in[0,T])

% ===== Frame 06 | PDF p.06 | 拡散過程の時間反転構造（導入） =====
  \begin{frame}{拡散モデルについて : 拡散過程の時間反転構造}
    \begin{equation}
      dX_t=-X_tdt+\sqrt{2}dW_t \quad (t\in[0,T])
    \end{equation}
    \begin{itemize}
      \item (2)の解を $\{X_t\}_{t\in[0,T]}$ とし, $t>0$ のとき, $X_t$ の密度を $p_t$ とする. このとき, 次の結果が知られている. 
    \end{itemize}
    \begin{theorem}[\cite{anderson82}]
      $\{Y_t\}_{t\in[0,T]}$ を次のSDEの解とする.
      \begin{equation}
        \left\{ \,
            \begin{aligned}
            & dY_t = (Y_t+\nabla \log{p_{T-t}(Y_t)})dt+\sqrt{2}dW_t \quad (0< t < T) \\
            & Y_0 \stackrel{\mathrm{d}}{=} X_T
            \end{aligned}
        \right.
      \end{equation}
      このとき, 任意の $t\in[0,T]$ に対して, $Y_t\stackrel{\mathrm{d}}{=}X_{T-t}$ が成り立つ.
    \end{theorem}
  \end{frame}

% ===== Frame 07 | PDF p.07 | 拡散過程の時間反転構造（2つのSDE） =====
  \begin{frame}{拡散モデルについて : 拡散過程の時間反転構造}
    \begin{itemize}
        \item $dX_t=-X_tdt+\sqrt{2}dW_t$ の解 $\{X_t\}_{t\in[0,T]}$ は, 
        \begin{equation*}
          X_t\stackrel{\mathrm{d}}{=}e^{-t}X_0+\sqrt{1-e^{-2t}}Z\quad (Z\sim \mathcal{N}(0,I_d), \quad Z\perp X_0)
        \end{equation*}
        となるので, 
        $T$ を十分大きく取ると $X_T\approx \mathcal{N}(0,I_d)$ となることが期待できる.
        \item $X_0\sim P_{\text{data}}$ とすると, $\{X_t\}_{t\in[0,T]}$ は $P_{\text{data}}$ から時間経過とともに $\mathcal{N}(0,I_d)$ へ近づいていく確率過程となる.
    \end{itemize}

% ===== Frame 09 | PDF p.09 | 3つの誤差の発生源 =====
    \begin{frame}{拡散モデルの収束解析について : 3つの誤差の発生源}
    \begin{equation*}
        \left\{ \,
            \begin{aligned}
            & dY_t = (Y_t+\nabla \log{p_{T-t}(Y_t)})dt+\sqrt{2}dB_t \quad (0\leqq t \leqq T) \\
            & Y_0 \stackrel{\mathrm{d}}{=} X_T\approx \mathcal{N}(0,I_d)
            \end{aligned}
        \right.
    \end{equation*}
      上記のSDEをシミュレーションすると, $Y_T\sim P_{\text{data}}$ からのサンプリングが得られる. しかし, 現実的には以下の3つの点が問題となる.
      \vspace{0.25cm}
      \begin{enumerate}
        \large
        \setlength{\itemsep}{0.4cm}
        \item 初期分布 $\mathcal{L}(X_T)$ からの厳密なサンプリングが得られない.
        \item スコア関数 $\nabla \log{p_{T-t}(Y_t)}$ が明示的に得られない.
        \item SDEを連続時間でシミュレーションできない.
      \end{enumerate}
  \end{frame}
% ===== Frame 10 | PDF p.10 | 3つの問題点への対応 =====
    \begin{frame}{拡散モデルの収束解析について : 3つの誤差の発生源}
    \begin{equation*}
        \left\{ \,
            \begin{aligned}
            & dY_t = (Y_t+\nabla \log{p_{T-t}(Y_t)})dt+\sqrt{2}dB_t \quad (0\leqq t \leqq T) \\
            & Y_0 \stackrel{\mathrm{d}}{=} X_T\approx \mathcal{N}(0,I_d)
            \end{aligned}
        \right.
    \end{equation*}
    \begin{equation*}
        \left\{ \,
            \begin{aligned}
            & d\hat{Y}_t = (\hat{Y}_{t_n}+\hat{s}(t_n,Y_{t_n}))dt+\sqrt{2}dB_t \quad (t_n\leqq t \leqq {t_{n+1}}) \\
            & \hat{Y}_0 \sim \mathcal{N}(0,I_d)
            \end{aligned}
        \right.
    \end{equation*}
    \begin{itemize}
      \item 前頁の3つの問題点への対応として, 次のようにしたSDE $\hat{Y}$ を考える.
      \begin{enumerate}
        \item 初期分布を, $\hat{Y}_0\sim \mathcal{N}(0,I_d)$ にする.
        \item スコア関数 $\nabla \log{p_{T-t}}$ を, 推定量 $\hat{s}(t,\cdot)$ に置き換える.
        \item 時間を離散化する. $(t_0=0\leqq t_1\leqq \dots \leqq t_N=T)$
      \end{enumerate}
    \end{itemize}
    \begin{itemize}
      \item 確率測度の近さを測る指標 $\rho$ を用いて, $\rho(P_{\text{data}},\mathcal{L}(\hat{Y}_T))=\rho(\mathcal{L}(Y_T),\mathcal{L}(\hat{Y}_T))$ を評価することが重要な課題である.
    \end{itemize}
  \end{frame}

% ===== Frame 12 | PDF p.12 | KLダイバージェンスの定義 =====
    \begin{frame}{拡散モデルの収束解析について : Kullback-Leibler divergence}
      \begin{definition}
        2つの確率測度 $P, Q$ のKullback-Leibler divergence $D_{\text{KL}}(P|Q)$ を, 次のように定義する.
        \begin{equation*}
          D_{\text{KL}}(P|Q):=        \left\{ \,
            \begin{aligned}
            & \mathbb{E}_P\left[\log{\frac{dP}{dQ}}\right] \quad \bigl(P\ll Q\bigr) \\
            & \infty \quad \quad \quad\quad \quad \quad(\text{otherwise})
            \end{aligned}
        \right.
        \end{equation*}
      \end{definition}
      Kullback-Leibler divergence は対称性を持たないため距離の公理を満たさないが, 確率測度の近さの指標としてよく用いられている.
  \end{frame}
% ===== Frame 13 | PDF p.13 | KL収束解析の先行研究 =====
    \begin{frame}{拡散モデルの収束解析について : Kullback-Leibler divergence}
      \begin{itemize}
        \setlength{\itemsep}{0.15em}
        \item KLでの収束解析は盛んに行われている. (\cite{benton_et_al_2024, chen_lee_lu_2023, chen_et_al_2023, koike_2026_follmer})
        \item  \cite{benton_et_al_2024} で得られた次の結果は, データの次元数 $d$ に関して最良の収束レートを得ている.
      \end{itemize}
      \vspace{-0.35em}
      \begin{theorem}[\cite{benton_et_al_2024}]
        ある定数 $C>0$ が存在して, 次の不等式が成立する.
        \begin{equation*}
          D_{\mathrm{KL}}\bigl(p_{\text{data}}\mid\mathcal{L}(\hat{Y}_{t_N})\bigr)
          \leqq C(\underbrace{\epsilon_{\text{score}}}_{\text{スコア推定誤差}}+\underbrace{\kappa^2dN+\kappa dT}_{\text{離散化誤差}}+\underbrace{de^{-2T}}_{\text{初期分布誤差}})
        \end{equation*}
        \vspace{-0.45em}
        \noindent\makebox[\linewidth][l]{%
          \textcolor{blue!70!black}{$\bullet$}\hspace{0.45em}%
          $t_0=0\leqq t_1\leqq\dots\leqq t_N=T$%
          \hspace{2em}
          \textcolor{blue!70!black}{$\bullet$}\hspace{0.45em}%
          $\displaystyle \kappa\leqq\max_{i=1,\dots,N}(t_i-t_{i-1})$%
        }
        \par\vspace{0.4em}
        \noindent
        \textcolor{blue!70!black}{$\bullet$}\hspace{0.45em}%
        $\displaystyle
          \epsilon_{\text{score}}
          =\sum_{i=1}^{N}(t_i-t_{i-1})
          \mathbb{E}\!\left[
            \left\|\nabla\log p_{t_i}(X_{t_i})
            -\hat{s}(t_i,X_{t_i})\right\|^2
          \right]$
      \end{theorem}
    \end{frame}
% ===== Frame 14 | PDF p.14 | 拡散モデルの収束解析について : Wasserstein distance =====
    \begin{frame}{拡散モデルの収束解析について : Wasserstein distance}
    \begin{definition}
      $\mathbb{R}^d$上の確率測度$P,Q$に対して, $\Pi(P,Q)=\{\pi\text{:}\mathbb{R}^d\times\mathbb{R}^d\text{上の確率測度}|\pi(\cdot,\mathbb{R}^d)=P,\pi(\mathbb{R}^d,\cdot)=Q\}$とする.
      このとき, $P,Q$の2-Wasserstein distance $W_2(P,Q)$を次のように定義する
      \begin{equation*}
        W_2(P,Q)=\inf_{\pi\in\Pi}\left(\int_{\mathbb{R}^d\times\mathbb{R}^d}|x-y|^2d\pi(x,y)\right)^{\frac{1}{2}}
      \end{equation*}
    \end{definition}
    \begin{itemize}
      \item Wasserstein distanceは距離の公理を満たす.
      \item 
    \end{itemize}
    \end{frame}
% ===== Frame 15 | PDF p.15 | Koike (2026) のWasserstein評価 =====
\begin{frame}{拡散モデルの収束解析について : Wasserstein distance}
  \small
  \begin{itemize}
    \setlength{\itemsep}{0.05em}
    \item \cite{koike_2026_wasserstein} は, 推定スコアの正則性のもとで鋭い評価を得た.
  \end{itemize}

  \vspace{-0.55em}

  \begin{theorem}[\cite{koike_2026_wasserstein}, Theorem 3.3]
    \noindent
    $\E[X_0]=0$, $\E[\|X_0\|_2^2]\leqq d$,
    $0<\alpha_0=e^{-2T}<\cdots<\alpha_N=1-\delta$,
    \par\vspace{0.05em}
    \noindent
    $h_i:=\alpha_{i+1}-\alpha_i
      \leqq\kappa\sqrt{1-\alpha_{i+1}}\ (i=0,\dots,N-1)$ とする.
    \par\vspace{0.1em}

    \vspace{-1.1em}
    \begin{align*}
      W_2\bigl(p_{\mathrm{data}},\mathcal{L}(\hat{Y}_{t_N})\bigr)
      \leqq{}& \sqrt{2}\exp\!\left(\int_{\alpha_0}^{1-\delta}L(s)\,ds\right)\Biggl[
      \\[-0.2em]
      &2\sqrt{\epsilon_{\mathrm{score}}}
      +\kappa\sqrt{d\left(1+\log\frac{1}{\delta}\right)}
      +\sqrt{2d}\,e^{-2T}\Biggr]
      +\sqrt{d(\delta+\delta^2)}.
    \end{align*}

    \vspace{-0.9em}
    {\footnotesize
      \centering
      誤差項は順に「スコア推定・離散化・初期分布・早期停止」に対応する.
      \par
    }

    \vspace{-0.65em}

    {\footnotesize
    \begin{align*}
      \epsilon_{\mathrm{score}}
      &:=
      \int_{\alpha_0}^{1-\delta}
      \frac{\varepsilon_{\mathrm{score}}(s)^2}{s}\,ds,
      \qquad
      q_\alpha:=\mathcal{L}\!\left(\sqrt{\alpha}X_0+\sqrt{1-\alpha}Z\right),
      \\[-0.2em]
      \varepsilon_{\mathrm{score}}(\alpha_i)^2
      &\geqq
      \E\!\left[\left\|\nabla\log q_{\alpha_i}(X_{\alpha_i})
      -\hat{s}(\alpha_i,X_{\alpha_i})\right\|_2^2\right],
      \qquad Z\sim\mathcal{N}(0,I_d).
    \end{align*}

    \vspace{-0.65em}
    \noindent 正則性条件：\quad
    $\displaystyle
      -\frac{2\alpha_i}{1-\alpha_i}I_d
      \preceq I_d+\nabla\hat{s}(\alpha_i,x)
      \preceq\alpha_iL(\alpha_i)I_d\quad\text{a.e. }x.$
    }
  \end{theorem}
\end{frame}
% ===== Frame 16 | PDF p.16 | データの内在次元に関する仮定 =====
  \begin{frame}{拡散モデルの収束解析について : データの内在次元}
    \begin{itemize}
        \item 現実世界のデータ分布はごく少数の変数で記述されると考えられている. これは多様体仮説と呼ばれ, 機械学習の成功の基盤と考えられている (\cite{pope_et_al_2021}).
        \item データ分布を記述する上で必要な変数の数を, データ分布の\textbf{内在次元 (intrinsic dimension)} と呼ぶ.
        \item \cite{pope_et_al_2021} は, 代表的な画像のデータセットに対する内在次元推定を行い, 画像は $224\times 224 \times 3 = 15028$ の画素値を持つが, 内在次元は $26\sim 43$ 程度であるという結果を得た.
    \end{itemize}
  \end{frame}

% ===== Frame 21 | PDF p.21 | 1ステップ誤差の計算（後半） =====

% ===== Frame 22 | PDF p.22 | 二次型離散Gronwall補題 =====
    \begin{frame}{証明の概要 : $\E[\|Y_{t_N}-\hat{Y}_{t_N}\|_2^2]^{\frac{1}{2}}$ の評価につなげるための補題}
      \begin{lemma}[\cite{arsenyan_vardanyan_dalalyan_2025}]
        非負数列 $(x_n)_{n\geqq0}$, $(B_n)_{n\geqq0}$,
        $(C_n)_{n\geqq0}$ と実数列 $(A_n)_{n\geqq0}$ が
        \begin{equation}
            x_{n+1}^2\leqq(e^{A_n}x_n+B_n)^2+C_n^2
        \end{equation}
        を満たすとする.
        このとき, 任意の $n\geqq0$ に対して
        \begin{align}
            x_{n+1}
            &\leqq
            \exp\left(\sum_{i=0}^{n}A_i\right)x_0
            +
            \sum_{j=0}^{n}
            \exp\left(\sum_{i=j+1}^{n}A_i\right)B_j
            \notag\\
            &\quad+
            \left\{
                \sum_{j=0}^{n}
                \exp\left(2\sum_{i=j+1}^{n}A_i\right)C_j^2
            \right\}^{\frac{1}{2}}
        \end{align}
        が成立する.
      \end{lemma}
    \end{frame}

% ===== Frame 23 | PDF p.23 | マルチンゲール性を使った誤差分解 =====
    \begin{frame}{証明の概要 : $\mu_{T-t}(Y_t)$ のマルチンゲール性を使った誤差分解}
        $\mu_{T-t}(Y_t)$ は $\mathcal{F}_t :=\sigma(Y_s,Z|0\leqq s\leqq t)$ に関してマルチンゲールである
        ことに注意すると, (23) 式の最初の3項の和を $S_n$, 最後の項を $R_n$ としたとき, $S_n$ は $\mathcal{F}_{t_n}$ 可測であり, $\E[R_n|\mathcal{F}_{t_n}]=0$ となることがわかる. 
        よって, 
        \begin{equation}
          \E[\langle S_n,R_n\rangle]=\E[\langle S_n,\E[R_n|\mathcal{F}_{t_n}]\rangle]=0
        \end{equation}
        となる. よって, 
        \begin{align}
          \E[\|Y_{t_{n+1}}-\hat{Y}_{t_{n+1}}\|_2^2]&=\E[\|S_n+R_n\|_2^2]\notag\\
          &=\E[\|S_n\|_2^2]+\E[\|R_n\|_2^2]+2\E[\langle S_n,R_n\rangle]\notag\\
          &=\E[\|S_n\|_2^2]+\E[\|R_n\|_2^2]
        \end{align}
    \end{frame}

% ===== Frame 24 | PDF p.24 | S_nの評価 =====

        S_n&=e^{h_n}(Y_{t_n}-\hat{Y}_{t_n})
        \notag\\
        &\quad+

% ===== Frame 27 | PDF p.27 | 初期分布誤差の評価 =====
    \begin{frame}{証明の概要：初期分布誤差を評価}
      初期値の取り方より
      \begin{equation}
        Y_0-\hat{Y}_0
        =e^{-T}X_0+(\sigma_T-1)Z.
      \end{equation}
      よって, $X_0$ と $Z$ が独立であることから,
      \begin{align}
        \E\left[\|Y_0-\hat{Y}_0\|_2^2\right]
        &\leqq
        e^{-2T}\E\left[\|X_0\|_2^2\right]
        +(1-\sigma_T)^2d.
      \end{align}
      また,
      \begin{equation}
        (1-\sigma_T)^2
        =
        \left(1-\sqrt{1-e^{-2T}}\right)^2
        \leqq e^{-4T}
      \end{equation}
      なので, 
      \begin{align}
        \E\left[\|Y_0-\hat{Y}_0\|_2^2\right]
        &\leqq
        e^{-2T}\E\left[\|X_0\|_2^2\right]
        +e^{-4T}d.
      \end{align}
    \end{frame}


% ===== Frame 28 | PDF p.28 | 離散化誤差の評価 =====
    \begin{frame}{証明の概要：離散化誤差を評価}
      \cite{huang_wei_chen_2026} より, 次の評価が得られる.
      \begin{equation}
        \sum_{n=0}^{N-1}\E\|R_n\|_2^2
        \leqq
        C\left\{
        \kappa\min\{\E\|X_0\|_2^2,Tk\log k\}
        +\kappa^2Nk\log k
        \right\}
      \end{equation}
      これは主に, 次の2つの補題から得られる.
      \begin{lemma}
        \begin{align}\label{eq:posterior-mean-increment}
          \E\left[
              \|\mu_{T-t}(Y_t)-\mu_{T-s}(Y_s)\|_2^2
          \right]=
          \E\left[Tr\left(\Sigma_{T-s}(Y_s)\right)\right]
          -
          \E\left[Tr\left(\Sigma_{T-t}(Y_t)\right)\right]
      \end{align}
      \end{lemma}
      \begin{lemma}
        ある定数 $C>0$ が存在して, 
        \begin{equation}\label{eq:posterior-covariance-bound}
          \E\left[\Tr\left(\Sigma_{T-t}(Y_t)\right)\right]
          \leqq
          C\min\left\{
              \E\|X_0\|_2^2,
              \frac{\sigma_{T-t}^2}{1-\sigma_{T-t}^2}k\log k
          \right\}
      \end{equation}
      \end{lemma}
    \end{frame}
